  <script>
    function updatePreview() {
      document.getElementById('previewName').innerText = document.getElementById('inputName').value || 'YOUR NAME';
      document.getElementById('previewDob').innerText = document.getElementById('inputDob').value || '--';
      document.getElementById('previewHeight').innerText = document.getElementById('inputHeight').value || '--';
      document.getElementById('previewEducation').innerText = document.getElementById('inputEducation').value || '--';
      document.getElementById('previewJob').innerText = document.getElementById('inputJob').value || '--';
      document.getElementById('previewContact').innerText = document.getElementById('inputContact').value || '--';
    }

    function downloadPDF(isPaid) {
      const watermark = document.getElementById('watermark');
      if (isPaid) {
        watermark.style.display = 'none'; // Hide watermark for paid version
      } else {
        watermark.style.display = 'block';
      }

      const element = document.getElementById('pdfTemplate');
      const opt = {
        margin:       0.5,
        filename:     'Marriage_Biodata_QuickDocs.pdf',
        image:        { type: 'jpeg', quality: 0.98 },
        html2canvas:  { scale: 2 },
        jsPDF:        { unit: 'in', format: 'letter', orientation: 'portrait' }
      };

      html2pdf().set(opt).from(element).save();
    }

    function triggerPayment() {
      alert("Payment Gateway Integration Point: Razorpay modal opens here for ₹29. On success, downloadPDF(true) triggers.");
    }
